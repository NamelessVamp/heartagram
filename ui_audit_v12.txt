export function readSidebarCollapsed(storage = window.localStorage) {
  try {
    return storage.getItem(STORAGE_KEY) === "true";
  } catch {
    return false;
  }
}

export function writeSidebarCollapsed(collapsed, storage = window.localStorage) {
  try {
    storage.setItem(STORAGE_KEY, String(Boolean(collapsed)));
  } catch {
    // Layout still works when storage is unavailable.
  }
}

function createNavigationButton(item, mobile = false) {
  const button = document.createElement("button");
  button.type = "button";
  button.className = mobile ? "mobile-nav-button" : "shell-nav-button";
  button.dataset.action = item.action;
  button.dataset.navScreen = item.screen;
  button.title = item.label;
  button.setAttribute("aria-label", item.label);
  button.dataset.tooltip = item.label;

  if (item.requiresProfile) {
    button.dataset.requiresProfile = "true";
  }

  const icon = document.createElement("span");
  icon.className = mobile ? "mobile-nav-icon" : "shell-nav-icon";
  icon.innerHTML = visualIcon(item.icon);
  icon.setAttribute("aria-hidden", "true");

  const label = document.createElement("span");
  label.className = mobile ? "mobile-nav-label" : "shell-nav-label";
  label.textContent = item.label;

  button.append(icon, label);
  return button;
}

function createSidebar() {
  const sidebar = document.createElement("aside");
  sidebar.id = "app-sidebar";
  sidebar.className = "app-sidebar";
  sidebar.setAttribute("aria-label", "Navegación de HEARTAGRAM");

  const brand = document.createElement("div");
  brand.className = "app-sidebar-brand";

  const mark = document.createElement("span");
  mark.className = "app-sidebar-brand-mark";
  mark.textContent = "H"; // HEARTAGRAM_VISUAL_SYSTEM_V1
  mark.setAttribute("aria-hidden", "true");

  const subtitle = document.createElement("span");
  subtitle.className = "app-sidebar-brand-subtitle";
  subtitle.textContent = "Heartagram";

  brand.append(mark, subtitle);

  const primary = document.createElement("nav");
  primary.className = "app-sidebar-nav";
  primary.setAttribute("aria-label", "Secciones principales");
  PRIMARY_NAV_ITEMS.forEach((item) => {
    primary.append(createNavigationButton(item));
  });

  const secondary = document.createElement("nav");
  secondary.className = "app-sidebar-secondary";
  secondary.setAttribute("aria-label", "Secciones secundarias");
  SECONDARY_NAV_ITEMS.forEach((item) => {
    secondary.append(createNavigationButton(item));
  });

  const toggle = document.createElement("button");
  toggle.type = "button";
  toggle.className = "sidebar-toggle";
  toggle.setAttribute("aria-controls", "app-sidebar");

  const toggleIcon = document.createElement("span");
  toggleIcon.className = "shell-nav-icon";
  toggleIcon.setAttribute("aria-hidden", "true");
  toggleIcon.dataset.visualToggleIcon = "collapse";
  toggleIcon.innerHTML = visualIcon("collapse");

  const toggleLabel = document.createElement("span");
  toggleLabel.className = "sidebar-toggle-label";
  toggleLabel.textContent = "Contraer";

  toggle.append(toggleIcon, toggleLabel);
  toggle.addEventListener("click", () => {
    const collapsed = !document.body.classList.contains("sidebar-collapsed");
    applySidebarCollapsed(collapsed);
    writeSidebarCollapsed(collapsed);
  });

  sidebar.append(brand, primary, secondary, toggle);
  return sidebar;
}

function createMobileNavigation() {
  const navigation = document.createElement("nav");
  navigation.id = "mobile-navigation";
  navigation.className = "mobile-navigation";
  navigation.setAttribute("aria-label", "Navegación principal móvil");
  PRIMARY_NAV_ITEMS.forEach((item) => {
    navigation.append(createNavigationButton(item, true));
  });
  return navigation;
}

function applySidebarCollapsed(collapsed) {
  document.body.classList.toggle("sidebar-collapsed", collapsed);
  const toggle = document.querySelector(".sidebar-toggle");
  if (!toggle) return;

  toggle.setAttribute("aria-expanded", String(!collapsed));
  toggle.setAttribute(
    "aria-label",
    collapsed ? "Expandir navegación" : "Contraer navegación",
  );

  const icon = toggle.querySelector(".shell-nav-icon");
  const label = toggle.querySelector(".sidebar-toggle-label");
  if (icon) {
    const name = collapsed ? "expand" : "collapse";
    icon.dataset.visualToggleIcon = name;
    icon.innerHTML = visualIcon(name);
  }
  if (label) label.textContent = collapsed ? "Expandir" : "Contraer";
}

export function ensureAppShell() {
  if (document.querySelector(".app-shell")) return;

  const app = document.getElementById("app");
  if (!app) throw new Error("No se encontró #app");

  const shell = document.createElement("div");
  shell.className = "app-shell";
  app.before(shell);
  shell.append(createSidebar(), app, createMobileNavigation());

  document.body.classList.add("has-app-shell");
  applySidebarCollapsed(readSidebarCollapsed());
}

export function updateAppShellNavigation(screenName, { hasProfile = false } = {}) {
  ensureAppShell();

  const navigationScreen = navigationScreenFor(screenName);
  const isQuiz = screenName === "quiz";
  document.body.classList.toggle("shell-focus-mode", isQuiz);

  for (const button of document.querySelectorAll("[data-nav-screen]")) {
    const requiresProfile = button.dataset.requiresProfile === "true";
    button.disabled = requiresProfile && !hasProfile;

    if (button.dataset.navScreen === navigationScreen) {
      button.setAttribute("aria-current", "page");
    } else {
      button.removeAttribute("aria-current");
    }
  }
}

export function formatCatalogDate(value) {
  if (typeof value !== "string" || !value.trim()) {
    return "Por confirmar";
  }

  const raw = value.trim();

  // Preserve year-only dates without inventing a month or day.
  if (/^\d{4}$/.test(raw)) {
    return raw;
  }

  // Only format complete ISO calendar dates.
  const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(raw);

  if (!match) {
    return raw;
  }

  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = Number(match[3]);

  const date = new Date(0);
  date.setUTCHours(0, 0, 0, 0);
  date.setUTCFullYear(year, month - 1, day);

  // Reject invalid dates instead of silently rolling them forward.
  if (
    date.getUTCFullYear() !== year ||
    date.getUTCMonth() !== month - 1 ||
    date.getUTCDate() !== day
  ) {
    return raw;
  }

  return new Intl.DateTimeFormat("es-MX", {
    day: "numeric",
    month: "long",
    year: "numeric",
    timeZone: "UTC",
  }).format(date);
}

function updateFavoriteButton(button, game) {
  const favoriteId = favoriteIdForCatalogGame(game);
  const selected = isFavorite(favoriteId);

  button.textContent = selected ? "♥ En mi lista" : "♡ Me interesa";
  button.setAttribute("aria-pressed", String(selected));
  button.setAttribute(
    "aria-label",
    selected
      ? `Quitar ${game.name} de mi lista`
      : `Agregar ${game.name} a mi lista`,
  );
}

function toggleCatalogFavorite(game, button) {
  toggleFavorite(favoriteIdForCatalogGame(game));
  updateFavoriteButton(button, game);
}

function updateOwnedButton(button, game) {
  const gameId = favoriteIdForCatalogGame(game);
  const owned = isOwnedGame(gameId);

  button.textContent = owned ? "✓ En Mi biblioteca" : "Ya lo tengo";
  button.setAttribute("aria-pressed", String(owned));
  button.setAttribute(
    "aria-label",
    owned
      ? `Quitar ${game.name} de Mi biblioteca`
      : `Agregar ${game.name} a Mi biblioteca`,
  );
}

function toggleCatalogOwned(game, favoriteButton, ownedButton) {
  const gameId = favoriteIdForCatalogGame(game);
  const nextOwned = !isOwnedGame(gameId);

  if (nextOwned && isFavorite(gameId)) {
    toggleFavorite(gameId);
  }

  setOwnedGame(gameId, nextOwned);
  updateFavoriteButton(favoriteButton, game);
  updateOwnedButton(ownedButton, game);
}

function hasActiveFilters() {
  return Boolean(
    state.query.trim() ||
    state.platform !== "ALL" ||
    state.genre !== "ALL" ||
    state.mode !== "ALL" ||
    state.sort !== "NAME"
  );
}

function renderActiveFilters() {
  const filters = [
    state.query.trim() ? ["query", `Buscar: ${state.query.trim()}`] : null,
    state.platform !== "ALL" ? ["platform", platformLabel(state.platform)] : null,
    state.genre !== "ALL" ? ["genre", catalogLabel(state.genre)] : null,
    state.mode !== "ALL" ? ["mode", catalogLabel(state.mode)] : null,
    state.sort !== "NAME"
      ? ["sort", state.sort === "RATING" ? "Mejor calificados" : "Más recientes"]
      : null,
  ].filter(Boolean);

  elements.activeFilters.replaceChildren();
  filters.forEach(([key, label]) => {
    const chip = createElement("button", "catalog-filter-chip", label);
    chip.type = "button";
    chip.dataset.clearCatalogFilter = key;
    chip.setAttribute("aria-label", `Quitar filtro ${label}`);
    chip.addEventListener("click", () => clearSingleFilter(key));
    elements.activeFilters.append(chip);
  });
  elements.clear.hidden = filters.length === 0;
}

function clearSingleFilter(key) {
  if (key === "query") {
    state.query = "";
    elements.query.value = "";
  } else if (key === "sort") {
    state.sort = "NAME";
    elements.sort.value = "NAME";
  } else if (["platform", "genre", "mode"].includes(key)) {
    state[key] = "ALL";
    elements[key].value = "ALL";
  }
  state.page = 1;
  renderResults();
}

function createCatalogCard(game) {
  const card = createElement("article", "catalog-card");

  const openButton = createElement("button", "catalog-card-open");

  openButton.type = "button";
  openButton.dataset.catalogId = String(game.nsuid);

  openButton.setAttribute("aria-label", `Ver detalles de ${game.name}`);

  const media = createElement("div", "catalog-card-media");

  const image = document.createElement("img");

  image.src = catalogImageUrl(game.cover, "cover_big");

  image.alt = `Portada de ${game.name}`;
  image.loading = "lazy";
  image.decoding = "async";

  image.addEventListener(
    "error",
    () => {
      image.remove();

      if (media.querySelector(".catalog-card-cover-fallback")) {
        return;
      }

      const initials = game.name
        .split(/\s+/)
        .filter(Boolean)
        .slice(0, 3)
        .map((word) => word[0])
        .join("")
        .toUpperCase();
      const fallback = createElement(
        "div",
        "catalog-card-cover-fallback",
        initials || "◇",
      );

      fallback.setAttribute("aria-hidden", "true");

      media.append(fallback);
    },
    {
      once: true,
    },
  );

  media.append(image);

  const body = createElement("div", "catalog-card-body");

  body.append(createElement("h3", "catalog-card-title", game.name));

  const metadata = [game.release_year, game.price_label, catalogLabel(game.game_type)].filter(
    Boolean,
  );

  if (metadata.length) {
    body.append(createElement("p", "catalog-card-meta", metadata.join(" · ")));
  }

  body.append(createBadges(game));

  openButton.append(media, body);

  const favoriteButton = createElement(
    "button",
    "favorite-button catalog-card-favorite",
    favoriteLabel(game),
  );

  favoriteButton.type = "button";
  favoriteButton.dataset.catalogFavorite = String(game.nsuid);

  updateFavoriteButton(favoriteButton, game);

  favoriteButton.addEventListener("click", () => {
    toggleCatalogFavorite(game, favoriteButton);
  });

  const ownedButton = createElement(
    "button",
    "library-button catalog-card-owned",
    "Ya lo tengo",
  );

  ownedButton.type = "button";
  ownedButton.dataset.catalogOwned = String(game.nsuid);
  updateOwnedButton(ownedButton, game);

  ownedButton.addEventListener("click", () => {
    toggleCatalogOwned(game, favoriteButton, ownedButton);
  });

  const actions = createElement("div", "catalog-card-actions");
  actions.append(favoriteButton, ownedButton);

  openButton.addEventListener("click", () => {
    openDetails(game);
  });

  card.append(openButton, actions);

  return card;
}

function renderResults() {
  const result = queryCatalog(state.catalog, {
    query: state.query,
    platform: state.platform,
    genre: state.genre,
    mode: state.mode,
    sort: state.sort,
    page: state.page,
    pageSize: PAGE_SIZE,
  });

  state.page = result.page;
  elements.grid.replaceChildren();

  if (result.total === 0) {
    elements.grid.append(createCatalogEmptyState());
  } else {
    const fragment = document.createDocumentFragment();

    result.games.forEach((game) => {
      fragment.append(createCatalogCard(game));
    });

    elements.grid.append(fragment);
  }

  renderActiveFilters();

  elements.count.textContent =
    result.total === 1
      ? "1 juego encontrado"
      : `${result.total.toLocaleString("es-MX")} juegos encontrados`;

  elements.page.textContent = `Página ${result.page} de ${result.pages}`;
  elements.previous.disabled = !result.hasPrevious;
  elements.next.disabled = !result.hasNext;
}

function renderGallery(details) {
  const gallery = elements.dialogGallery;
  gallery.replaceChildren();

  const sources = [
    details.cover?.url_2x,
    ...(details.screenshots ?? []),
    ...(details.artworks ?? []),
  ]
    .filter((source) => typeof source === "string" && source.trim().length > 0)
    .map((source) => catalogImageUrl(source, "screenshot_big"))
    .filter(Boolean)
    .slice(0, 6);

  const message = createElement(
    "p",
    "catalog-gallery-status",
    sources.length
      ? "Cargando imágenes..."
      : "No hay imágenes disponibles para este juego.",
  );

  message.setAttribute("role", "status");
  gallery.append(message);

  if (sources.length === 0) {
    return;
  }

  let pending = sources.length;
  let loaded = 0;

  function updateStatus() {
    pending -= 1;

    if (loaded > 0) {
      message.remove();
    } else if (pending === 0) {
      message.textContent =
        "No se pudieron cargar las imágenes. Los detalles del juego siguen disponibles.";
    }
  }

  sources.forEach((source, index) => {
    const image = document.createElement("img");

    image.alt = `Imagen ${index + 1} de ${details.name}`;
    image.decoding = "async";
    image.hidden = true;

    image.addEventListener(
      "load",
      () => {
        loaded += 1;
        image.hidden = false;
        updateStatus();
      },
      { once: true },
    );

    image.addEventListener(
      "error",
      () => {
        image.remove();
        updateStatus();
      },
      { once: true },
    );

    gallery.append(image);
    image.src = source;
  });
}

function priceAmount(value) {
  if (value === null || value === undefined || value === "") return null;
  const parsed = Number(value);
  return Number.isFinite(parsed) ? parsed : null;
}
function formatPrice(value, currency = "MXN") {
  const amount = priceAmount(value);
  if (amount === null) return "Precio no disponible";
  return new Intl.NumberFormat("es-MX", { style: "currency", currency, maximumFractionDigits: 2 }).format(amount);
}

function renderNintendoDetails(details) {
  const isNintendo = details.catalog_source === "nintendo";
  elements.dialogNintendo.hidden = !isNintendo;
  if (!isNintendo) return;
  const currency = details.price_currency || "MXN";
  const regular = priceAmount(details.regular_price);
  const sale = priceAmount(details.sale_price);
  elements.dialogPrice.textContent = formatPrice(sale ?? regular, currency);
  const hasSale = sale !== null && regular !== null && sale < regular;
  elements.dialogSale.hidden = !hasSale;
  elements.dialogSale.textContent = hasSale ? `Antes ${formatPrice(regular, currency)} · ${details.percent_off || 0}% de descuento` : "";
  elements.dialogRating.textContent = details.content_rating?.label || "No disponible";
  elements.dialogPlayers.textContent = details.player_count || "No disponible";
  elements.dialogWays.textContent = textOrFallback(details.ways_to_play);
  elements.dialogAvailability.textContent = textOrFallback(details.availability);
  fillDetailChips(elements.dialogContentDescriptorList, elements.dialogContentDescriptors, details.content_descriptors);
  fillDetailChips(elements.dialogFeatureList, elements.dialogFeatures, [...(details.features ?? []), ...(details.nso_features ?? [])]);
  elements.dialogDemo.hidden = !details.demo_nsuid;
}
function textOrFallback(values, fallback = "No disponible") {
  if (Array.isArray(values)) {
    return values.length ? values.join(", ") : fallback;
  }

  return values || fallback;
}

function renderDetails(indexGame, details) {
  state.selectedGame = indexGame;
  const useNintendoDetail = isNintendoDetail(details, indexGame);
  const legacyCopy = elements.dialogBody.querySelector(".catalog-dialog-copy");
  elements.nintendoProductDetail.hidden = !useNintendoDetail;
  if (legacyCopy) legacyCopy.hidden = useNintendoDetail;
  elements.dialogGallery.hidden = useNintendoDetail;
  if (elements.dialogNintendo) elements.dialogNintendo.hidden = true;
  if (useNintendoDetail) {
    renderNintendoProductDetail(elements.nintendoProductDetail, details, indexGame);
    return;
  }

  state.selectedGame = indexGame;

  const platforms = (details.platforms ?? [])
    .map((platform) => typeof platform === "string" ? platformLabel(platform) : platform.name)
    .filter(Boolean);

  elements.dialogPlatforms.textContent = textOrFallback(platforms);
  elements.dialogTitle.textContent = details.name;
  elements.dialogSummary.textContent =
    details.summary || "Nintendo Store Mexico no tiene una descripcion disponible.";

  const hasStory = Boolean(details.storyline);
  elements.dialogStorySection.hidden = !hasStory;
  elements.dialogStory.textContent = details.storyline || "";

  elements.dialogRelease.textContent = formatCatalogDate(details.release_date);
  elements.dialogDevelopers.textContent = textOrFallback(details.developers);
  elements.dialogPublishers.textContent = textOrFallback(details.publishers);
  elements.dialogModes.textContent = textOrFallback(
    (details.game_modes ?? []).map(catalogLabel),
  );


  renderDialogBadges(details);
  renderGallery(details);
  updateFavoriteButton(elements.dialogFavorite, indexGame);
  updateOwnedButton(elements.dialogOwned, indexGame);
}

async function openDetails(game) {
  dialogReturnFocus = document.activeElement instanceof HTMLElement ? document.activeElement : null;
  elements.dialogLoading.hidden = false;
  elements.dialogLoading.textContent = "Cargando detalles...";
  elements.dialogBody.hidden = true;

  if (!elements.dialog.open) {
    elements.dialog.showModal();
  }

  try {
    const details = await loadGameDetails(state.catalog, game);
    renderDetails(game, details);
    elements.dialogLoading.hidden = true;
    elements.dialogBody.hidden = false;
  } catch (error) {
    elements.dialogLoading.textContent =
      "No se pudieron cargar los detalles de este juego.";
    console.error(error);
  }
}

function bindEvents() {
  elements.query.addEventListener("input", () => {
    state.query = elements.query.value;
    state.page = 1;
    renderResults();
  });

  elements.platform.addEventListener("change", () => {
    state.platform = elements.platform.value;
    state.page = 1;
    renderResults();
  });

  elements.genre.addEventListener("change", () => {
    state.genre = elements.genre.value;
    state.page = 1;
    renderResults();
  });

  elements.mode.addEventListener("change", () => {
    state.mode = elements.mode.value;
    state.page = 1;
    renderResults();
  });

  elements.sort.addEventListener("change", () => {
    state.sort = elements.sort.value;
    state.page = 1;
    renderResults();
  });

  elements.clear.addEventListener("click", resetFilters);
  elements.retry.addEventListener("click", () => {
    initialized = false;
    void initializeCatalogExplorer();
  });

  elements.previous.addEventListener("click", () => {
    if (state.page <= 1) return;
    state.page -= 1;
    renderResults();
    elements.grid.scrollIntoView({ behavior: matchMedia("(prefers-reduced-motion: reduce)").matches ? "auto" : "smooth", block: "start" });
  });

  elements.next.addEventListener("click", () => {
    state.page += 1;
    renderResults();
    elements.grid.scrollIntoView({ behavior: matchMedia("(prefers-reduced-motion: reduce)").matches ? "auto" : "smooth", block: "start" });
  });

  elements.dialogClose.addEventListener("click", closeDialog);
  elements.dialog.addEventListener("close", restoreDialogFocus);

  elements.dialog.addEventListener("click", (event) => {
    if (event.target === elements.dialog) {
      closeDialog();
    }
  });

  elements.dialogFavorite.addEventListener("click", () => {
    if (!state.selectedGame) return;
    toggleCatalogFavorite(state.selectedGame, elements.dialogFavorite);
    renderResults();
  });

  elements.dialogOwned.addEventListener("click", () => {
    if (!state.selectedGame) return;
    toggleCatalogOwned(
      state.selectedGame,
      elements.dialogFavorite,
      elements.dialogOwned,
    );
    renderResults();
  });

  window.addEventListener("heartagram:favorites-changed", () => {
    if (state.catalog) {
      renderResults();
    }
  });

  window.addEventListener("heartagram:library-changed", () => {
    if (state.catalog) {
      renderResults();
    }
  });

function normalizeText(value) {

export function favoriteIdForCatalogGame(gameOrId) {
  if (gameOrId && typeof gameOrId === "object") {
    if (typeof gameOrId.catalog_id === "string") return gameOrId.catalog_id;
    if (gameOrId.nsuid) return `nintendo:MX:${String(gameOrId.nsuid)}`;

export function isCatalogFavoriteId(value) {
  return typeof value === "string" && /^nintendo:MX:\d{10,20}$/.test(value);

  return isCatalogFavoriteId(value) ? value.slice("nintendo:MX:".length) : null;
}

export function splitFavoriteIds(values = []) {
  const curatedIds = [];
  const catalogIds = [];
  for (const value of values) {
    const catalogId = catalogIdFromFavoriteId(value);
    if (catalogId === null) curatedIds.push(value);
    else catalogIds.push(catalogId);

export function findCatalogGamesByIds(catalog, catalogIds) {

  const byId = new Map(catalog.games.map((game) => [String(game.nsuid), game]));

export function getCatalogFacets(catalog) {
  const genres = new Set(), modes = new Set(), platforms = new Set(), types = new Set();
  for (const game of catalog.games) {
    game.genres?.forEach((value) => genres.add(value));
    game.game_modes?.forEach((value) => modes.add(value));
    game.platforms?.forEach((value) => platforms.add(value));
    if (game.game_type) types.add(game.game_type);
  }
  const sort = (values) => [...values].sort((a, b) => a.localeCompare(b, "es-MX", { sensitivity: "base" }));
  return { genres: sort(genres), modes: sort(modes), platforms: sort(platforms), types: sort(types) };
}

function matchesValue(values, selected) {
  return !selected || selected === "ALL" || (Array.isArray(values) && values.includes(selected));
}

function compareGames(left, right, sort) {
  if (sort === "RATING") {
    const difference = (right.rating ?? -1) - (left.rating ?? -1);
    if (difference) return difference;
  }
  if (sort === "NEWEST") {
    const difference = (right.release_year ?? -1) - (left.release_year ?? -1);
    if (difference) return difference;
  }
  return left.name.localeCompare(right.name, "es-MX", { sensitivity: "base", numeric: true });
}

export function queryCatalog(catalog, {
  query = "", platform = "ALL", genre = "ALL", mode = "ALL",
  gameType = "ALL", sort = "NAME", page = 1, pageSize = DEFAULT_PAGE_SIZE,
} = {}) {
  const safePageSize = Math.max(1, Number(pageSize) || DEFAULT_PAGE_SIZE);
  const normalizedQuery = normalizeText(query);
  const filtered = catalog.games
    .filter((game) => !normalizedQuery || normalizeText(game.name).includes(normalizedQuery))
    .filter((game) => matchesValue(game.platforms, platform))
    .filter((game) => matchesValue(game.genres, genre))
    .filter((game) => matchesValue(game.game_modes, mode))
    .filter((game) => !gameType || gameType === "ALL" || game.game_type === gameType)
    .sort((left, right) => compareGames(left, right, sort));
  const total = filtered.length;
  const pages = Math.max(1, Math.ceil(total / safePageSize));
  const safePage = Math.min(Math.max(1, Number(page) || 1), pages);
  const start = (safePage - 1) * safePageSize;
  return {
    games: filtered.slice(start, start + safePageSize), total, page: safePage,
    pages, pageSize: safePageSize, hasPrevious: safePage > 1, hasNext: safePage < pages,
  };
}

export async function loadGameDetails(_catalog, game) {
  if (!game || game.catalog_source !== "nintendo") {
    throw new TypeError("El producto no pertenece al catalogo Nintendo.");
  }
  return { ...game, cover: game.cover_detail, platforms: game.platform_details };
}

function normalizeIds(value) {

          return;
        }

        state.index += 1;
        renderCurrentQuestion();
      }, NEXT_QUESTION_DELAY_MS);
    },
  );

    (game) => !ownedSet.has(game.id),

    catalogLoading = false,
    catalogError = false,
  } = {}) => {
    const interestedCatalogGames = catalogGames.filter((game) => {

      return favoriteSet.has(gameId) && !ownedSet.has(gameId);
    });

    const ownedCatalogGames = catalogGames.filter((game) =>
      ownedSet.has(favoriteIdForCatalogGame(game)),
    );

    ui.renderWishlist({
      interestedCuratedGames,
      interestedCatalogGames,
      ownedCuratedGames,
      ownedCatalogGames,
      catalogLoading,
      catalogError,

      onOpenCatalogGame: openCatalogGameDetails,

      onRemoveCatalogGame: removeCatalogFavorite,

      onMoveCuratedToLibrary: (game) => moveGameToLibrary(game.id),

      onRemoveCuratedFromLibrary: (game) => removeGameFromLibrary(game.id),

      onMoveCatalogToLibrary: moveCatalogGameToLibrary,

      onRemoveCatalogFromLibrary: removeCatalogGameFromLibrary,
    });
  };

  render({
    catalogLoading: catalogIds.length > 0,
  });

  if (catalogIds.length === 0) {
    return;
  }

  try {
    const catalogGames = await resolveCatalogFavorites(requestedIds);

    render({
      catalogGames,
    });
  } catch (error) {
    render({
      catalogError: true,
    });

function numericPrice(value) {
  if (value === null || value === undefined || value === "") return null;
  if (typeof value === "number") return Number.isFinite(value) ? value : null;

  if (typeof value === "object") {
    for (const key of ["amount", "value", "price", "raw_value", "finalPrice", "regularPrice"]) {
      const parsed = numericPrice(value[key]);
      if (parsed !== null) return parsed;
    }
    return null;
  }

  const normalized = String(value)
    .trim()
    .replace(/[^0-9,.-]/g, "")
    .replace(/,(?=\d{3}(?:\D|$))/g, "")
    .replace(",", ".");

  if (!normalized) return null;
  const parsed = Number(normalized);
  return Number.isFinite(parsed) ? parsed : null;
}

function priceLabel(product) {
  const value = numericPrice(product.sale_price) ?? numericPrice(product.regular_price);
  if (value === null) return null;
  return new Intl.NumberFormat("es-MX", {
    style: "currency",
    currency: product.price_currency || "MXN",
    maximumFractionDigits: 2,
  }).format(value);
}

}

  window.localStorage.setItem(PROFILE_KEY, JSON.stringify(profile));

export function getSavedProfile() {
  try {
    const value = window.localStorage.getItem(PROFILE_KEY);

    if (!value) return null;

    const profile = JSON.parse(value);

    if (
      typeof profile !== "object" ||
      profile === null ||
      !profile.vector ||
      !profile.archetypes
    ) {
      return null;
    }

    return profile;
  } catch {
    return null;
  }
}

  return getSavedProfile() !== null;
}

export function clearSavedProfile() {
  window.localStorage.removeItem(PROFILE_KEY);

// Turns quiz answers into a player profile: a 6-dimension vector,
// ranked archetypes and the special fields used by the recommender.

export const DIMENSIONS = ["pace", "social", "structure", "session", "mood", "challenge"];

const ARCHETYPE_ORDER = [
  "EXPLORER", "COMPETITOR", "COLLECTOR", "CREATOR",
  "SURVIVOR", "STRATEGIST", "STORY_SEEKER", "HORROR_FAN",
];

const DEFAULT_HORROR_TOLERANCE = 1;

/**
 * Per-dimension bounds reachable through the quiz.
 * Needed so every dimension normalizes to the same -2..+2 scale,
 * regardless of how many questions touch it.
 */
function dimensionBounds(questions) {
  const bounds = Object.fromEntries(DIMENSIONS.map((d) => [d, { max: 0, min: 0 }]));
  for (const question of questions) {
    for (const dim of DIMENSIONS) {
      const values = question.options.map((o) => o.dims[dim] ?? 0);
      bounds[dim].max += Math.max(0, ...values);
      bounds[dim].min += Math.min(0, ...values);
    }
  }
  return bounds;
}

function normalize(sum, bound) {
  if (sum >= 0) return bound.max === 0 ? 0 : (2 * sum) / bound.max;
  return bound.min === 0 ? 0 : (2 * sum) / Math.abs(bound.min);
}

function rankArchetypes(scores, support) {
  return ARCHETYPE_ORDER
    .map((id) => ({ id, score: scores[id] ?? 0, support: support[id] ?? 0 }))
    .sort((a, b) => b.score - a.score || b.support - a.support);
}

/**
 * @param {Array} questions  parsed questions.json
 * @param {Object} answers   { q1: "a", q2: "c", ... }
 */
export function computeProfile(questions, answers) {
  const sums = Object.fromEntries(DIMENSIONS.map((d) => [d, 0]));
  const archScores = {};
  const archSupport = {};
  const vibes = new Set();
  let horrorTolerance = DEFAULT_HORROR_TOLERANCE;
  let couplePref = null;

  for (const question of questions) {
    const option = question.options.find((o) => o.id === answers[question.id]);
    if (!option) throw new Error(`Missing answer for ${question.id}`);

    for (const [dim, value] of Object.entries(option.dims)) sums[dim] += value;
    for (const [arch, value] of Object.entries(option.arch)) {
      archScores[arch] = (archScores[arch] ?? 0) + value;
      archSupport[arch] = (archSupport[arch] ?? 0) + 1;
    }
    option.vibes.forEach((v) => vibes.add(v));

    if (question.special === "horror_tolerance") horrorTolerance = option.value;
    if (question.special === "couple_pref") couplePref = option.couple;
  }

  const bounds = dimensionBounds(questions);
  const vector = Object.fromEntries(DIMENSIONS.map((d) => [d, normalize(sums[d], bounds[d])]));

  const ranking = rankArchetypes(archScores, archSupport);
  const [first, second] = ranking;
  const tiedPrimary = first.score === second.score && first.support === second.support;

  return {
    vector,
    archetypes: { primary: first.id, secondary: second.id, tiedPrimary, ranking },
    horrorTolerance,
    couplePref,
    vibes: [...vibes],
  };
}

// Ranks verified games against a player profile.
// Pure functions only: no DOM access.

import { DIMENSIONS } from "./profile.js";

const WEIGHTS = {
  ARCHETYPE_PRIMARY_BONUS: 0.10,
  ARCHETYPE_SECONDARY_BONUS: 0.05,
  VIBES_BONUS: 0.10,
  VIBES_BONUS_CAP: 0.20,
  HORROR_PENALTY_PER_LEVEL: 0.25,
  COUPLE_PREF_BONUS: 0.05,
};

const SECTION_LIMITS = { FOR_YOU: 3, MIGHT_HOOK: 7 };

// Social is half-weighted here because co-op has its own section (buildDuo).
// Without this, liking multiplayer pushes great solo games out of the top 3.
const DIMENSION_WEIGHTS = { social: 0.5 };

function toArray(vector) {
  return DIMENSIONS.map((d) => (vector[d] ?? 0) * (DIMENSION_WEIGHTS[d] ?? 1));
}

function cosine(a, b) {
  const dot = a.reduce((sum, value, i) => sum + value * b[i], 0);
  const normA = Math.hypot(...a);
  const normB = Math.hypot(...b);
  return normA === 0 || normB === 0 ? 0 : dot / (normA * normB);
}

export function scoreGame(profile, game) {
  let score = cosine(toArray(profile.vector), toArray(game.vector));

  if (game.personality.includes(profile.archetypes.primary)) score += WEIGHTS.ARCHETYPE_PRIMARY_BONUS;
  if (game.personality.includes(profile.archetypes.secondary)) score += WEIGHTS.ARCHETYPE_SECONDARY_BONUS;

  const sharedVibes = game.vibes.filter((v) => profile.vibes.includes(v)).length;
  score += Math.min(sharedVibes * WEIGHTS.VIBES_BONUS, WEIGHTS.VIBES_BONUS_CAP);

  // Penalized, not removed: a curious player can still discover it further down.
  const excessHorror = game.horror_level - profile.horrorTolerance;
  if (excessHorror > 0) score -= excessHorror * WEIGHTS.HORROR_PENALTY_PER_LEVEL;

  return score;
}

export function rankGames(profile, games) {
  return games
    .filter((game) => game.status === "VERIFICADO")
    .map((game) => ({ game, score: scoreGame(profile, game) }))
    .sort((a, b) => b.score - a.score);
}

/** Picks a strong game from the opposite side of the user's mood axis. */
function pickOutOfZone(profile, ranked, excludedIds) {
  const userMood = profile.vector.mood || ranked[0]?.game.vector.mood || 0;
  const opposite = (game) => Math.sign(game.vector.mood) === -Math.sign(userMood) && userMood !== 0;
  return ranked.find(({ game }) => !excludedIds.has(game.id) && opposite(game)) ?? null;
}

export function buildSections(profile, ranked) {
  const forYou = ranked.slice(0, SECTION_LIMITS.FOR_YOU);
  const mightHook = ranked.slice(SECTION_LIMITS.FOR_YOU, SECTION_LIMITS.MIGHT_HOOK);
  const shown = new Set([...forYou, ...mightHook].map(({ game }) => game.id));
  return { forYou, mightHook, outOfZone: pickOutOfZone(profile, ranked, shown) };
}

export function duoBadges(game) {
  const m = game.multiplayer;
  const badges = [];
  if (m.online_coop === true || m.online_versus === true) badges.push("ONLINE");
  if (m.local_wireless === true) badges.push("MISMO CUARTO");
  if (m.same_console === true) badges.push("UNA CONSOLA");
  return badges;
}

/**
 * Games for playing together. Order: Inti already owns it, then games that
 * work both remotely and in person, then profile score.
 */
export function buildDuo(profile, ranked) {
  return ranked
    .map(({ game, score }) => {
      const badges = duoBadges(game);
      const bonus = game.couple.includes(profile.couplePref) ? WEIGHTS.COUPLE_PREF_BONUS : 0;
      return { game, score: score + bonus, badges };
    })
    .filter(({ badges }) => badges.length > 0)
    .sort((a, b) => {
      const owned = Number(b.game.couple.includes("INTI_LIBRARY_MATCH")) - Number(a.game.couple.includes("INTI_LIBRARY_MATCH"));
      const bothWays = (x) => Number(x.badges.includes("ONLINE") && x.badges.length > 1);
      return owned || bothWays(b) - bothWays(a) || b.score - a.score;
    });
}

    button.type = "button";

    button.tabIndex = option.id === selectedId || (!selectedId && optionIndex === 0) ? 0 : -1;

    const number = el("span", "quiz-option-index", String(optionIndex + 1).padStart(2, "0"));
    number.setAttribute("aria-hidden", "true");
    const copy = el("span", "quiz-option-copy", option.text);

    button.addEventListener("click", () => {
      container.querySelectorAll(".quiz-option").forEach((candidate) => {
        candidate.setAttribute("aria-checked", "false");
        candidate.tabIndex = -1;
      });
      button.setAttribute("aria-checked", "true");
      button.tabIndex = 0;
      onSelect(option.id);
    });
    container.append(button);
  });

  container.onkeydown = (event) => {
    if (!["ArrowDown", "ArrowRight", "ArrowUp", "ArrowLeft"].includes(event.key)) return;
    event.preventDefault();
    const options = [...container.querySelectorAll(".quiz-option")];
    const activeIndex = Math.max(0, options.indexOf(document.activeElement));
    const direction = ["ArrowDown", "ArrowRight"].includes(event.key) ? 1 : -1;
    const target = options[(activeIndex + direction + options.length) % options.length];
    options.forEach((option) => { option.tabIndex = -1; });
    target.tabIndex = 0;
    target.focus();
  };
}

function drawRadar(canvas, vector) {
  const styles = getComputedStyle(document.documentElement);
  const color = (name) => styles.getPropertyValue(name).trim();
  const ratio = window.devicePixelRatio || 1;

  canvas.width = RADAR.SIZE * ratio;
  canvas.height = RADAR.SIZE * ratio;
  const ctx = canvas.getContext("2d");
  ctx.scale(ratio, ratio);

  const center = RADAR.SIZE / 2;
  const angle = (i) => (Math.PI * 2 * i) / DIMENSIONS.length - Math.PI / 2;
  const point = (i, r) => [
    center + Math.cos(angle(i)) * r,
    center + Math.sin(angle(i)) * r,
  ];

  ctx.strokeStyle = color("--color-border");
  ctx.lineWidth = 1;
  for (let ring = 1; ring <= RADAR.RINGS; ring++) {
    ctx.beginPath();
    DIMENSIONS.forEach((_, i) =>
      ctx.lineTo(...point(i, (RADAR.RADIUS * ring) / RADAR.RINGS)),
    );
    ctx.closePath();
    ctx.stroke();
  }

  ctx.fillStyle = color("--color-text-muted");
  ctx.font = "12px system-ui, sans-serif";
  ctx.textAlign = "center";
  ctx.textBaseline = "middle";
  DIMENSIONS.forEach((dim, i) =>
    ctx.fillText(DIMENSION_LABELS[dim], ...point(i, RADAR.RADIUS + 22)),
  );

  // Vector values are -2..+2; map to 0..1 of the radius.
  ctx.beginPath();
  DIMENSIONS.forEach((dim, i) =>
    ctx.lineTo(...point(i, (((vector[dim] ?? 0) + 2) / 4) * RADAR.RADIUS)),
  );
  ctx.closePath();
  ctx.fillStyle = "rgba(224, 98, 122, 0.25)";
  ctx.strokeStyle = color("--color-accent-strong");
  ctx.lineWidth = 2;
  ctx.fill();
  ctx.stroke();
}

export function gameCard(game, options = {}) {

    .toUpperCase();

    card.append(badges);
  }

  const actions = el("div", "library-card-actions");

  if (showFavorite && !libraryMode) {
    const favoriteButton = el("button", "favorite-button", favoriteLabel(game.id));
    favoriteButton.type = "button";
    favoriteButton.dataset.favoriteId = game.id;
    favoriteButton.setAttribute("aria-label", favoriteAriaLabel(game));
    favoriteButton.setAttribute("aria-pressed", String(isFavorite(game.id)));
    favoriteButton.addEventListener("click", () => {
      const selected = toggleFavorite(game.id);
      favoriteButton.textContent = selected ? "♥ En mi lista" : "♡ Me interesa";
      favoriteButton.setAttribute("aria-pressed", String(selected));
      favoriteButton.setAttribute("aria-label", favoriteAriaLabel(game));
    });

  }

  if (!libraryMode && typeof onMoveToLibrary === "function") {
    const ownedButton = el("button", "library-button", "Ya lo tengo");

    ownedButton.addEventListener("click", () => onMoveToLibrary(game));
    actions.append(ownedButton);

  if (libraryMode && typeof onRemoveFromLibrary === "function") {
    const removeButton = el(
      "button",
      "library-button library-button-remove",
      "Quitar de mi biblioteca",
    );
    removeButton.type = "button";
    removeButton.setAttribute("aria-label", `Quitar ${game.title} de Mi biblioteca`);
    removeButton.addEventListener("click", () => onRemoveFromLibrary(game));
    actions.append(removeButton);
  }

  if (actions.children.length > 0) card.append(actions);
  return card;
}

  } else if (catalogLoading && ownedCuratedGames.length === 0) {
    libraryContent.append(el("p", "catalog-status", "Cargando Mi biblioteca..."));
  }

  container.append(librarySection);

  const counts = {
    interested: interestedCuratedGames.length + interestedCatalogGames.length,
    owned: ownedCuratedGames.length + ownedCatalogGames.length,
  };

  window.dispatchEvent(
    new CustomEvent("heartagram:collection-rendered", {
      detail: { counts },
    }),
  );
}

function decorateCollectionPanel(panel, type) {
  if (!panel) return;
  panel.classList.toggle("collection-panel-interested", type === "interested");
  panel.classList.toggle("collection-panel-owned", type === "owned");

  panel.querySelectorAll(".card, .catalog-wishlist-card").forEach((card) => {
    card.dataset.collectionType = type;
    if (type === "owned" && !card.querySelector(".collection-owned-seal")) {
      const seal = document.createElement("span");
      seal.className = "collection-owned-seal";
      seal.textContent = "EN TU BIBLIOTECA";
      card.prepend(seal);
    }
  });

  const empty = panel.querySelector(".empty");
  if (!empty || empty.dataset.collectionEnhanced === "true") return;
  empty.dataset.collectionEnhanced = "true";
  empty.classList.add("collection-empty-state");
  empty.replaceChildren();

  const mark = document.createElement("span");
  mark.className = "collection-empty-mark";
  mark.setAttribute("aria-hidden", "true");
  mark.textContent = type === "owned" ? "◇" : "♡";

  const title = document.createElement("h3");
  title.className = "collection-empty-title";
  title.textContent = type === "owned" ? "Tu biblioteca está lista para crecer" : "Todavía no has apartado ningún juego";

  const copy = document.createElement("p");
  copy.className = "collection-empty-copy";
  copy.textContent = type === "owned"
    ? "Marca Ya lo tengo desde Explorar o mueve aquí un juego de Me interesa."
    : "Explora el catálogo y guarda juegos para revisarlos después.";

  const action = document.createElement("button");
  action.type = "button";
  action.className = "btn btn-primary";
  action.dataset.action = "to-catalog";
  action.textContent = "Explorar juegos";

  empty.append(mark, title, copy, action);
}

window.addEventListener("heartagram:collection-rendered", () => {
  decorateCollectionPanel(document.getElementById("collection-panel-interested"), "interested");
  decorateCollectionPanel(document.getElementById("collection-panel-owned"), "owned");
});

// HEARTAGRAM Visual System V1: global atmosphere and ceremonial empty states.


function installAtmosphere() {
  if (!document.querySelector('.visual-atmosphere')) {
    const layer = document.createElement('div');
    layer.className = 'visual-atmosphere';
    layer.setAttribute('aria-hidden', 'true');
    document.body.prepend(layer);
  }
  if (!document.querySelector('.visual-corner-mark')) {
    const mark = document.createElement('div');
    mark.className = 'visual-corner-mark';
    mark.textContent = 'HEARTAGRAM // FOR ADRI // LOCAL SIGNAL';
    mark.setAttribute('aria-hidden', 'true');
    document.body.append(mark);
  }
}

function decorateHome() {
  const header = document.querySelector('.home-header');
  if (!header || header.querySelector('.visual-home-sigil')) return;
  const note = document.createElement('p');
  note.className = 'visual-home-note';
  note.textContent = 'made for player two ✦';
  const orbit = document.createElement('div');
  orbit.className = 'visual-home-orbit';
  orbit.innerHTML = '<span class="visual-orbit-ring visual-orbit-ring-a"></span><span class="visual-orbit-ring visual-orbit-ring-b"></span><span class="visual-orbit-core">♡</span><span class="visual-orbit-star">✦</span>';
  orbit.setAttribute('aria-hidden', 'true');
  header.append(note, orbit);
}

function decorateScreenLabels() {
  const labels = [
    ['.discover-header', 'DISCOVERY // FOR ADRI'],
    ['.catalog-screen-header', 'GAME ARCHIVE // SWITCH'],
    ['.collection-header', 'SAVED SIGNALS // YOUR LIST'],
  ];
  for (const [selector, text] of labels) {
    const host = document.querySelector(selector);
    if (!host || host.querySelector('.visual-module-label')) continue;
    const label = document.createElement('p');
    label.className = 'visual-module-label';
    label.textContent = text;
    host.prepend(label);
  }
}

function installVisualSystem() {
  installAtmosphere();
  decorateHome();
  decorateScreenLabels();
}

if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', installVisualSystem, { once: true });
} else {
  installVisualSystem();
}
